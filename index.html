/**
 * ECE Register Bit-Masking in C
 * Author: Yashwant (18, ECE)
 * Focus: Embedded Pin Control & Bit Manipulation
 */

#include <stdio.h>
#include <stdint.h>

// Define virtual microcontroller port and control pin
#define PIN_TRANSMIT    (1 << 3)   // 4th bit: RF Transmit Enable
#define PIN_CLOCK       (1 << 0)   // 1st bit: Synchronous Clock

void display_register(uint8_t reg) {
    printf("Current Register State: 0x%02X\n", reg);
}

int main(void) {
    uint8_t gpio_port = 0x00;  // Initial state: All low
    
    // 1. Turn ON RF Transmit bit without modifying other pins
    gpio_port |= PIN_TRANSMIT;
    printf("[*] Transmit Bit Set.\n");
    display_register(gpio_port);

    // 2. Toggle Synchronous Clock using XOR
    gpio_port ^= PIN_CLOCK;
    printf("[*] Clock Toggled High.\n");
    display_register(gpio_port);

    // 3. Safely Clear Transmit Bit
    gpio_port &= ~PIN_TRANSMIT;
    printf("[*] Transmit Bit Cleared.\n");
    display_register(gpio_port);

    return 0;
}
